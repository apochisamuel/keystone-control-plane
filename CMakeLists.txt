cmake_minimum_required(VERSION 3.20)
project(aegis_control_plane VERSION 0.1.0 LANGUAGES CXX)

option(AEGIS_BUILD_TESTS "Build tests" ON)
option(AEGIS_BUILD_TOOLS "Build tools" ON)
option(AEGIS_BUILD_FUZZERS "Build libFuzzer harnesses" OFF)
option(AEGIS_ENABLE_SANITIZERS "Enable address and undefined sanitizers" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB AEGIS_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_library(aegis_control_plane STATIC ${AEGIS_SOURCES})
target_include_directories(aegis_control_plane PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(aegis_control_plane PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(aegis_control_plane PRIVATE -Wall -Wextra -Wpedantic)

if(AEGIS_ENABLE_SANITIZERS)
  target_compile_options(aegis_control_plane PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(aegis_control_plane PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

if(AEGIS_BUILD_TESTS)
  enable_testing()
  add_executable(aegis_tests tests/test_main.cpp)
  target_link_libraries(aegis_tests PRIVATE aegis_control_plane)
  add_test(NAME aegis_tests COMMAND aegis_tests)
endif()

if(AEGIS_BUILD_TOOLS)
  add_executable(aegis-inspect tools/inspect_capture.cc)
  target_link_libraries(aegis-inspect PRIVATE aegis_control_plane)
  add_executable(aegis-generate-example tools/generate_examples.cc)
  target_link_libraries(aegis-generate-example PRIVATE aegis_control_plane)
  add_executable(aegis-policy-check tools/policy_check.cc)
  target_link_libraries(aegis-policy-check PRIVATE aegis_control_plane)
  add_executable(aegis-recover tools/recover.cc)
  target_link_libraries(aegis-recover PRIVATE aegis_control_plane)
endif()

if(AEGIS_BUILD_FUZZERS)
  foreach(target capture_stream_fuzzer frame_sequence_fuzzer policy_blob_fuzzer extension_block_fuzzer)
    add_executable(${target} fuzz/${target}.cc)
    target_link_libraries(${target} PRIVATE aegis_control_plane)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  endforeach()
endif()
